Quick Start

改一行 base_url
即刻接入。

如果你已经在用 OpenAI / Anthropic 官方 SDK，几乎不需要改业务代码。 鉴权头、流式协议、工具调用全保持一致。

  • 不锁定你的客户端选型
  • cURL / Python / Node / Go 全跑得通
  • 支持 system prompt 缓存，节省重复读 token
# 兼容 OpenAI 协议端点
curl https://xuanji.dev/v1/chat/completions \
  -H "Authorization: Bearer $XUANJI_KEY" \
  -H "Content-Type: application/json" \
  -d '{
    "model": "claude-sonnet-4.5",
    "stream": true,
    "messages": [
      {"role":"user","content":"你好，玄机"}
    ]
  }'
from openai import OpenAI

client = OpenAI(
    api_key=os.environ["XUANJI_KEY"],
    base_url="https://xuanji.dev/v1",
)

stream = client.chat.completions.create(
    model="claude-sonnet-4.5",
    messages=[{"role": "user", "content": "你好，玄机"}],
    stream=True,
)
for chunk in stream:
    print(chunk.choices[0].delta.content or "", end="")
import anthropic

client = anthropic.Anthropic(
    api_key=os.environ["XUANJI_KEY"],
    base_url="https://xuanji.dev",  # 同一把 Key
)

with client.messages.stream(
    model="claude-sonnet-4.5",
    max_tokens=2048,
    messages=[{"role": "user", "content": "用 python 算 fib(30)"}],
    tools=[{"name": "python", "input_schema": {"type": "object"}}],
) as stream:
    for text in stream.text_stream:
        print(text, end="", flush=True)
